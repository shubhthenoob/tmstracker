🚀 QUICK START (5 minutes to working cron job)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1️⃣ READ: README_CRON.md (this is the starting point)

2️⃣ SETUP ENVIRONMENT VARIABLES in Vercel:
   • CRON_SECRET = openssl rand -base64 32
   • DATABASE_URL = your Neon connection string
   • GOOGLE_SHEETS_API_KEY = from Google Cloud Console
   • GOOGLE_SHEETS_ID = from your Google Sheet URL
   • GOOGLE_SHEETS_RANGE = Sheet1!A2:G1000

3️⃣ CREATE DATABASE TABLES:
   Run the SQL from scripts/add-sync-logs.sql in Neon dashboard

4️⃣ DEPLOY:
   git add .
   git commit -m "Add Google Sheets cron sync"
   git push

5️⃣ VERIFY:
   Open /admin/sync-status and click "Trigger Sync Now"

Google Sheets
    ↓
Vercel Cron (11 AM & 11 PM UTC)
    ↓
/api/sync-sheet endpoint
    ├─ Authenticate (CRON_SECRET)
    ├─ Fetch Google Sheets data
    ├─ Sync to PostgreSQL (upsert)
    └─ Log operation
    ↓
PostgreSQL Database (Neon)
    ├─ tasks table (synced data)
    └─ sync_logs table (audit trail)
    ↓
Admin Dashboard (/admin/sync-status)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

tasks table:
├─ id (Primary Key)
├─ date, task, assignee, hours
├─ type, status
├─ Timestamps (created_at, updated_at, last_synced_at)
└─ Indexes on: date, assignee, status

sync_logs table:
├─ id (Primary Key)
├─ sync_time, status (success/failed/pending)
├─ rows_synced, error_message, duration_ms
├─ created_at
└─ Indexes on: sync_time, status
